SELECT 'PURCHASE' AS movement, pi.id AS ref_id, pi.date, pi.quantity FROM sma_purchase_items pi WHERE pi.product_id = 30 AND pi.warehouse_id = 1 UNION ALL SELECT 'SALE' AS movement, si.id AS ref_id, s.date, -si.quantity AS quantity FROM sma_sale_items si JOIN sma_sales s ON s.id = si.sale_id WHERE si.product_id = 30 AND si.warehouse_id = 1 AND s.sale_status = 'completed' UNION ALL SELECT CONCAT('ADJUSTMENT ', ai.type) AS movement, ai.id AS ref_id, a.date, CASE WHEN ai.type = 'addition' THEN ai.quantity ELSE -ai.quantity END AS quantity FROM sma_adjustment_items ai JOIN sma_adjustments a ON a.id = ai.adjustment_id WHERE ai.product_id = 30 AND ai.warehouse_id = 1 ORDER BY date, ref_id;
